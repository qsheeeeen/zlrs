cmake_minimum_required(VERSION 3.20.0)
get_filename_component(zlrs_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CONF_FILE "${zlrs_root}/app/prj.conf" "${zlrs_root}/app/debug.conf"
 "${zlrs_root}/app/trace-debug.conf" "${CMAKE_CURRENT_SOURCE_DIR}/prj.conf")
set(DTC_OVERLAY_FILE "${zlrs_root}/app/debug.overlay")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(latency LANGUAGES C)
include("${zlrs_root}/app/CMakeLists.txt")
get_target_property(trace_sources zephyr SOURCES)
list(FILTER trace_sources EXCLUDE REGEX "/tracing_backend_ram\\.c$")
set_property(TARGET zephyr PROPERTY SOURCES "${trace_sources}")
target_sources(app PRIVATE latency.c "${zlrs_root}/tests/support/rolling_trace_ram.c")
foreach(symbol zlrs_link_tx_space zlrs_serial_read zlrs_serial_receive zlrs_slot_timer_launch_at)
 target_link_options(app PUBLIC "-Wl,--wrap=${symbol}")
endforeach()
